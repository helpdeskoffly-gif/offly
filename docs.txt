create view public.community_post_stats as
select
  cp.id as post_id,
  cp.user_id,
  u.username,
  u.full_name,
  u.avatar_url,
  cp.content,
  cp.title,
  cp.content_type,
  cp.source_type,
  cp.source_id,
  cp.hashtags,
  cp.is_public,
  cp.created_at,
  cp.updated_at,
  COALESCE(like_counts.like_count, 0::bigint) as like_count,
  0 as comment_count
from
  community_posts cp
  join users u on cp.user_id = u.id
  left join (
    select
      community_likes.post_id,
      count(*) as like_count
    from
      community_likes
    group by
      community_likes.post_id
  ) like_counts on cp.id = like_counts.post_id;

  create table public.community_posts (

  is_public boolean null default true,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint community_posts_pkey primary key (id),
  constraint community_posts_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE,
  constraint community_posts_content_type_check check (
    (
      content_type = any (
        array[
          'text'::text,
          'image'::text,
          'anti_todo'::text,
          'checkin'::text
        ]
      )
    )
  ),
  constraint community_posts_source_type_check check (
    (
      source_type = any (
        array[
          'anti_todo'::text,

          'manual'::text
        ]
      )
    )
  )
