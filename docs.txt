-- Community Post Stats View - Optimized view for displaying posts with like/comment counts
create or replace view community_post_stats as
select 
    cp.id as post_id,
    cp.user_id,
    u.username,
    u.full_name,
    u.avatar_url,
    cp.content,
    cp.title,
    cp.content_type,
    cp.source_type,
    cp.source_id,
    cp.hashtags,
    cp.is_public,
    cp.created_at,
    cp.updated_at,
    coalesce(like_counts.like_count, 0) as like_count,
    coalesce(comment_counts.comment_count, 0) as comment_count
from community_posts cp
join users u on cp.user_id = u.id
left join (
    select post_id, count(*) as like_count
    from community_likes
    group by post_id
) like_counts on cp.id = like_counts.post_id
left join (
    select post_id, count(*) as comment_count
    from community_comments
    group by post_id
) comment_counts on cp.id = comment_counts.post_id;

-- Function to award points to a user
create or replace function award_user_points(
  p_user_id uuid,
  p_amount integer,
  p_source_type text,
  p_source_id uuid default null,
  p_description text default null
)
returns jsonb as $$
declare
  v_new_total integer;
begin
  -- Insert or update user points
  insert into user_points (user_id, total_points)
  values (p_user_id, p_amount)
  on conflict (user_id)
  do update set 
    total_points = user_points.total_points + p_amount,
    updated_at = now()
  returning total_points into v_new_total;

  -- Record transaction
  insert into point_transactions (user_id, amount, transaction_type, source_type, source_id, description)
  values (p_user_id, p_amount, 'earned', p_source_type, p_source_id, p_description);

  return jsonb_build_object('success', true, 'new_total', v_new_total, 'amount_awarded', p_amount);
exception
  when others then
    return jsonb_build_object('success', false, 'error', sqlerrm);
end;
$$ language plpgsql;

-- Function to spend user points
create or replace function spend_user_points(
  p_user_id uuid,
  p_amount integer,
  p_source_type text,
  p_source_id uuid default null,
  p_description text default null
)
returns jsonb as $$
declare
  v_current_points integer;
  v_new_total integer;
begin
  -- Check current points
  select total_points into v_current_points
  from user_points
  where user_id = p_user_id;

  if v_current_points is null or v_current_points < p_amount then
    return jsonb_build_object('success', false, 'error', 'Insufficient points');
  end if;

  -- Spend points
  update user_points
  set 
    total_points = total_points - p_amount,
    updated_at = now()
  where user_id = p_user_id
  returning total_points into v_new_total;

  -- Record transaction
  insert into point_transactions (user_id, amount, transaction_type, source_type, source_id, description)
  values (p_user_id, p_amount, 'spent', p_source_type, p_source_id, p_description);

  return jsonb_build_object('success', true, 'new_total', v_new_total, 'amount_spent', p_amount);
exception
  when others then
    return jsonb_build_object('success', false, 'error', sqlerrm);
end;
$$ language plpgsql;

-- Function to get personalized feed based on user preferences and hobbies
create or replace function get_personalized_feed(
    p_user_id uuid,
    p_limit integer default 20,
    p_offset integer default 0
)
returns table (
    post_id uuid,
    user_id uuid,
    username text,
    full_name text,
    avatar_url text,
    content text,
    title text,
    content_type text,
    source_type text,
    source_id uuid,
    hashtags text[],
    is_public boolean,
    created_at timestamptz,
    updated_at timestamptz,
    like_count bigint,
    comment_count bigint,
    preference_score integer
) as $$
begin
    return query
    select distinct
        cp.id as post_id,
        cp.user_id,
        u.username,
        u.full_name,
        u.avatar_url,
        cp.content,
        cp.title,
        cp.content_type,
        cp.source_type,
        cp.source_id,
        cp.hashtags,
        cp.is_public,
        cp.created_at,
        cp.updated_at,
        coalesce(stats.like_count, 0) as like_count,
        coalesce(stats.comment_count, 0) as comment_count,
        case 
            -- calculate preference match score based on multiple factors
            when current_user_prefs.preferred_hashtags is null then 0
            else (
                -- score based on hashtag overlap (0-50 points)
                (
                    select count(*) * 10
                    from unnest(cp.hashtags) as post_tag
                    where post_tag = any(current_user_prefs.preferred_hashtags)
                    limit 5
                ) +
                -- score based on content type match (0-30 points)
                case 
                    when cp.content_type = any(current_user_prefs.preferred_content_types) then 30
                    else 0
                end +
                -- score based on post author's hobbies matching current user's hobbies (0-20 points)
                (
                    select count(*) * 5
                    from unnest(post_author.hobbies) as author_hobby
                    join unnest(current_user.hobbies) as user_hobby on author_hobby = user_hobby
                    limit 4
                )
            )
        end as preference_score
    from community_posts cp
    join users u on cp.user_id = u.id
    left join users post_author on cp.user_id = post_author.id
    left join users current_user on current_user.id = p_user_id
    left join user_preferences current_user_prefs on current_user_prefs.user_id = p_user_id
    left join (
        select 
            cp2.id as post_id,
            count(distinct cl.id) as like_count,
            count(distinct cc.id) as comment_count
        from community_posts cp2
        left join community_likes cl on cp2.id = cl.post_id
        left join community_comments cc on cp2.id = cc.post_id
        group by cp2.id
    ) stats on cp.id = stats.post_id
    where 
        cp.is_public = true
        and cp.user_id != p_user_id  -- don't show user's own posts
        and cp.created_at >= now() - interval '30 days'  -- only recent posts (last 30 days)
    order by 
        preference_score desc,  -- first by preference match
        cp.created_at desc      -- then by recency
    limit p_limit
    offset p_offset;
end;
$$ language plpgsql;

-- Function to get friends feed - posts from users you follow
create or replace function get_friends_feed(
    p_user_id uuid,
    p_limit integer default 20,
    p_offset integer default 0
)
returns table (
    post_id uuid,
    user_id uuid,
    username text,
    full_name text,
    avatar_url text,
    content text,
    title text,
    content_type text,
    source_type text,
    source_id uuid,
    hashtags text[],
    is_public boolean,
    created_at timestamptz,
    updated_at timestamptz,
    like_count bigint,
    comment_count bigint
) as $$
begin
    return query
    select 
        cp.id as post_id,
        cp.user_id,
        u.username,
        u.full_name,
        u.avatar_url,
        cp.content,
        cp.title,
        cp.content_type,
        cp.source_type,
        cp.source_id,
        cp.hashtags,
        cp.is_public,
        cp.created_at,
        cp.updated_at,
        coalesce(stats.like_count, 0) as like_count,
        coalesce(stats.comment_count, 0) as comment_count
    from community_posts cp
    join users u on cp.user_id = u.id
    join user_relationships ur on (ur.following_id = cp.user_id and ur.follower_id = p_user_id)
    left join (
        select 
            cp2.id as post_id,
            count(distinct cl.id) as like_count,
            count(distinct cc.id) as comment_count
        from community_posts cp2
        left join community_likes cl on cp2.id = cl.post_id
        left join community_comments cc on cp2.id = cc.post_id
        group by cp2.id
    ) stats on cp.id = stats.post_id
    where 
        cp.is_public = true
        and ur.status = 'accepted'
        and cp.created_at >= now() - interval '30 days'  -- only recent posts (last 30 days)
    order by 
        cp.created_at desc
    limit p_limit
    offset p_offset;
end;
$$ language plpgsql;

-- Function to get suggested friends - shows all users but ordered by preference
create or replace function get_suggested_friends(
    p_user_id uuid,
    p_limit integer default 10
)
returns table (
    user_id uuid,
    username text,
    full_name text,
    avatar_url text,
    bio text,
    shared_hobbies text[],
    hobby_match_count integer
) as $$
declare
    current_user_hobbies text[];
begin
    -- get the current user's hobbies first
    select hobbies into current_user_hobbies 
    from users 
    where id = p_user_id;
    
    -- return all users but ordered by preference (hobby matches first)
    return query
    select 
        u.id as user_id,
        u.username,
        u.full_name,
        u.avatar_url,
        u.bio,
        case 
            when current_user_hobbies is not null and u.hobbies is not null then
                (
                    select array_agg(shared_hobby)
                    from unnest(u.hobbies) as shared_hobby
                    where shared_hobby = any(current_user_hobbies)
                )
            else null
        end as shared_hobbies,
        case 
            when current_user_hobbies is not null and u.hobbies is not null then
                (
                    select count(*)::integer
                    from unnest(u.hobbies) as user_hobby
                    where user_hobby = any(current_user_hobbies)
                )
            else 0
        end as hobby_match_count
    from users u
    left join user_relationships ur on (
        (ur.follower_id = p_user_id and ur.following_id = u.id) or
        (ur.follower_id = u.id and ur.following_id = p_user_id)
    )
    where 
        u.id != p_user_id  -- don't show yourself
        and ur.id is null  -- not already connected
    order by 
        -- first: users with hobby matches (highest count first)
        hobby_match_count desc,
        -- then: users with any hobbies (even if no match)
        case when u.hobbies is not null and array_length(u.hobbies, 1) > 0 then 1 else 0 end desc,
        -- finally: all other users by creation date (newest first)
        u.created_at desc
    limit p_limit;
end;
$$ language plpgsql;
